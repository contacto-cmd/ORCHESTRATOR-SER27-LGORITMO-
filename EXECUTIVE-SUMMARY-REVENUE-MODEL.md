```javascript
// cloudflare-worker.js
export default {
  async fetch(request, env) {
    // 1. Validate JWT token
    const token = request.headers.get('Authorization');
    if (!token) {
      return new Response('Unauthorized', { status: 401 });
    }
    
    // 2. Verify token (using ORCHESTRATOR public key)
    const verified = await verifyJWT(token, env.PUBLIC_KEY);
    if (!verified) {
      return new Response('Invalid token', { status: 401 });
    }
    
    // 3. Route to correct backend
    const url = new URL(request.url);
    
    if (url.pathname.startsWith('/api/jobs')) {
      // Route to ORCHESTRATOR
      return fetch('https://ser27-orchestrator.railway.app' + url.pathname, {
        method: request.method,
        headers: request.headers,
        body: request.body,
      });
    }
    
    if (url.pathname.startsWith('/api/chat')) {
      // Route to SINI Backend
      return fetch('https://sini-backend-ser27.railway.app' + url.pathname, {
        method: request.method,
        headers: request.headers,
        body: request.body,
      });
    }
    
    if (url.pathname === '/') {
      // Route to Vercel frontend
      return fetch('https://royal-hub.vercel.app' + url.pathname);
    }
    
    return new Response('Not found', { status: 404 });
  }
};
```

```javascript
// src/api.js (Vercel frontend)
const ORCHESTRATOR_URL = process.env.REACT_APP_ORCHESTRATOR_URL;

export async function submitJob(command) {
  const response = await fetch(`${ORCHESTRATOR_URL}/api/jobs/submit`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${getToken()}`,
    },
    body: JSON.stringify({ command }),
  });
  
  return response.json();
}

export async function getJobStatus(jobId) {
  const response = await fetch(
    `${ORCHESTRATOR_URL}/api/jobs/${jobId}`,
    {
      headers: {
        'Authorization': `Bearer ${getToken()}`,
      },
    }
  );
  
  return response.json();
}

// Usage in React component
function Dashboard() {
  const [jobs, setJobs] = useState([]);
  
  useEffect(() => {
    const interval = setInterval(async () => {
      const result = await fetch(`${ORCHESTRATOR_URL}/api/jobs/pending`);
      const data = await result.json();
      setJobs(data.jobs);
    }, 2000); // Refresh every 2 seconds
    
    return () => clearInterval(interval);
  }, []);
  
  return (
    <div>
      <h1>Job Dashboard</h1>
      {jobs.map(job => (
        <div key={job.id}>
          <p>{job.command}</p>
          <p>Status: {job.status}</p>
        </div>
      ))}
    </div>
  );
}
```
